## Vue.js 集成示例

```javascript
import request from '@/utils/request'

export function getKeywordRankingAnalysis(params) {
```

```vue
<script>
import { getKeywordRankingAnalysis } from '@/api/keywordRankingAnalysis'
import SummaryCard from './components/SummaryCard.vue'
import MatrixDisplay from './components/MatrixDisplay.vue'
import StatisticsCharts from './components/StatisticsCharts.vue'
import ComparisonAnalysis from './components/ComparisonAnalysis.vue'

export default {
  name: 'KeywordRankingAnalysis',
  components: {
    SummaryCard,
    MatrixDisplay,
    StatisticsCharts,
    ComparisonAnalysis
  },
  data() {
    return {
      loading: false,
      queryForm: {
        city: '',
        rankType: null,
        keywords: [],
        maxPosition: 50,
        aggregationType: 'latest',
        displayMode: 'single',
        comparisonType: 'position'
      },
      dateRange: [],
      analysisData: null
    }
  },
  methods: {
    async handleQuery() {
      if (!this.queryForm.city || !this.queryForm.rankType) {
        this.$message.error('请填写必填字段')
        return
      }

      this.loading = true
      try {
        const params = {
          ...this.queryForm,
          startDate: this.dateRange?.[0],
          endDate: this.dateRange?.[1]
        }
        
        const response = await getKeywordRankingAnalysis(params)
        this.analysisData = response.data
        this.$message.success('分析完成')
      } catch (error) {
        this.$message.error('查询失败：' + error.message)
      } finally {
        this.loading = false
      }
    },
    
    handleReset() {
      this.queryForm = {
        city: '',
        rankType: null,
        keywords: [],
        maxPosition: 50,
        aggregationType: 'latest',
        displayMode: 'single',
        comparisonType: 'position'
      }
      this.dateRange = []
      this.analysisData = null
    }
  }
}
</script>
```

```vue
<script>
export default {
  name: 'MatrixTable',
  props: {
    keywordMatrix: {
      type: Object,
      required: true
    },
    headers: {
      type: Array,
      required: true
    }
  },
  computed: {
    dateHeaders() {
      return this.headers.slice(1) // 去掉第一个"位置"列
    }
  },
  methods: {
    getAsinClass(asin) {
      // 根据ASIN生成不同的颜色类
      const hash = this.hashCode(asin)
      return `asin-color-${Math.abs(hash) % 8}`
    },
    
    hashCode(str) {
      let hash = 0
      for (let i = 0; i < str.length; i++) {
        const char = str.charCodeAt(i)
        hash = ((hash << 5) - hash) + char
        hash = hash & hash // Convert to 32bit integer
      }
      return hash
    }
  }
}
</script>
```

```vue
<!-- components/StatisticsCharts.vue -->
<template>
  <el-row :gutter="20" class="statistics-charts">
    <!-- ASIN频次图表 -->
    <el-col :span="12">
      <el-card>
        <template #header>ASIN出现频次</template>
        <div ref="asinFrequencyChart" style="height: 300px;"></div>
      </el-card>
    </el-col>
    
    <!-- 位置稳定性图表 -->
    <el-col :span="12">
      <el-card>
        <template #header>位置稳定性</template>
        <div ref="stabilityChart" style="height: 300px;"></div>
      </el-card>
    </el-col>
    
    <!-- 关键词覆盖度 -->
    <el-col :span="24" style="margin-top: 20px;">
      <el-card>
        <template #header>关键词覆盖度</template>
        <el-table :data="coverageTableData" style="width: 100%">
          <el-table-column prop="keyword" label="关键词" />
          <el-table-column prop="coveredPositions" label="覆盖位置数" />
          <el-table-column prop="totalPositions" label="总位置数" />
          <el-table-column prop="coverageRate" label="覆盖率">
            <template #default="{ row }">
              <el-progress 
                :percentage="Math.round(row.coverageRate * 100)"
                :color="getProgressColor(row.coverageRate)"
              />
            </template>
          </el-table-column>
        </el-table>
      </el-card>
    </el-col>
  </el-row>
</template>

<script>
import * as echarts from 'echarts'

export default {
  name: 'StatisticsCharts',
  props: {
    statistics: {
      type: Object,
      required: true
    }
  },
  computed: {
    coverageTableData() {
      if (!this.statistics.keywordCoverage) return []
      return Object.values(this.statistics.keywordCoverage)
    }
  },
  mounted() {
    this.$nextTick(() => {
      this.initCharts()
    })
  },
  watch: {
    statistics: {
      handler() {
        this.$nextTick(() => {
          this.initCharts()
        })
      },
      deep: true
    }
  },
  methods: {
    initCharts() {
      this.initAsinFrequencyChart()
      this.initStabilityChart()
    },
    
    initAsinFrequencyChart() {
      const chart = echarts.init(this.$refs.asinFrequencyChart)
      const data = Object.entries(this.statistics.asinFrequency || {})
        .map(([asin, count]) => ({ name: asin, value: count }))
        .sort((a, b) => b.value - a.value)
        .slice(0, 10) // 只显示前10个
      
      const option = {
        title: {
          text: 'Top 10 ASIN',
          left: 'center'
        },
        tooltip: {
          trigger: 'item',
          formatter: '{b}: {c} ({d}%)'
        },
        series: [{
          type: 'pie',
          radius: '70%',
          data: data,
          emphasis: {
            itemStyle: {
              shadowBlur: 10,
              shadowOffsetX: 0,
              shadowColor: 'rgba(0, 0, 0, 0.5)'
            }
          }
        }]
      }
      
      chart.setOption(option)
    },
    
    initStabilityChart() {
      const chart = echarts.init(this.$refs.stabilityChart)
      const data = Object.entries(this.statistics.positionStability || {})
        .sort((a, b) => b[1] - a[1])
        .slice(0, 20) // 只显示前20个位置
      
      const option = {
        title: {
          text: '位置稳定性评分',
          left: 'center'
        },
        tooltip: {
          trigger: 'axis',
          formatter: '{b}: {c}'
        },
        xAxis: {
          type: 'category',
          data: data.map(item => item[0]),
          axisLabel: {
            rotate: 45
          }
        },
        yAxis: {
          type: 'value',
          min: 0,
          max: 1
        },
        series: [{
          type: 'bar',
          data: data.map(item => item[1]),
          itemStyle: {
            color: function(params) {
              const value = params.value
              if (value >= 0.8) return '#67C23A'
              if (value >= 0.6) return '#E6A23C'
              return '#F56C6C'
            }
          }
        }]
      }
      
      chart.setOption(option)
    },
    
    getProgressColor(rate) {
      if (rate >= 0.8) return '#67C23A'
      if (rate >= 0.6) return '#E6A23C'
      return '#F56C6C'
    }
  }
}
</script>
```

## React.js 集成示例

### 1. 主组件 (React + Ant Design)

```jsx
import React, { useState, useEffect } from 'react'
import { 
  Card, Form, Input, Select, DatePicker, Radio, Button, 
  message, Tabs, Table, Progress, Row, Col 
} from 'antd'
import { getKeywordRankingAnalysis } from '@/api/keywordRankingAnalysis'
import * as echarts from 'echarts'

const { RangePicker } = DatePicker
const { Option } = Select

const KeywordRankingAnalysis = () => {
  const [form] = Form.useForm()
  const [loading, setLoading] = useState(false)
  const [analysisData, setAnalysisData] = useState(null)

  const handleSubmit = async (values) => {
    setLoading(true)
    try {
      const params = {
        ...values,
        startDate: values.dateRange?.[0]?.format('YYYY-MM-DD'),
        endDate: values.dateRange?.[1]?.format('YYYY-MM-DD')
      }
      delete params.dateRange
      
      const response = await getKeywordRankingAnalysis(params)
      setAnalysisData(response.data)
      message.success('分析完成')
    } catch (error) {
      message.error('查询失败：' + error.message)
    } finally {
      setLoading(false)
    }
  }

  const renderMatrix = (keywordMatrix, headers) => {
    const columns = [
      {
        title: '位置',
        dataIndex: 'position',
        key: 'position',
        fixed: 'left',
        width: 100,
        render: (text, record) => (
          <span style={{ 
            background: '#f0f2f5', 
            padding: '2px 8px', 
            borderRadius: '4px',
            fontSize: '12px',
            fontWeight: 'bold'
          }}>
            {text}
          </span>
        )
      },
      ...headers.slice(1).map((header, index) => ({
        title: header,
        dataIndex: ['data', index],
        key: header,
        width: 120,
        render: (text) => (
          <div style={{ textAlign: 'center' }}>
            {text ? (
              <span style={{
                background: getAsinColor(text),
                color: 'white',
                padding: '4px 8px',
                borderRadius: '4px',
                fontSize: '12px',
                fontWeight: 'bold'
              }}>
                {text}
              </span>
            ) : (
              <span style={{ color: '#ccc' }}>-</span>
            )}
          </div>
        )
      }))
    ]

    return (
      <Table
        columns={columns}
        dataSource={keywordMatrix.rows}
        rowKey="position"
        scroll={{ x: 'max-content', y: 600 }}
        pagination={false}
        size="small"
      />
    )
  }

  const getAsinColor = (asin) => {
    const colors = [
      '#409EFF', '#67C23A', '#E6A23C', '#F56C6C', 
      '#909399', '#9C27B0', '#FF9800', '#4CAF50'
    ]
    let hash = 0
    for (let i = 0; i < asin.length; i++) {
      hash = ((hash << 5) - hash) + asin.charCodeAt(i)
      hash = hash & hash
    }
    return colors[Math.abs(hash) % colors.length]
  }

  return (
    <div className="keyword-ranking-analysis">
      {/* 查询表单 */}
      <Card title="查询条件" style={{ marginBottom: 20 }}>
        <Form
          form={form}
          layout="inline"
          onFinish={handleSubmit}
          initialValues={{
            maxPosition: 50,
            aggregationType: 'latest',
            displayMode: 'single'
          }}
        >
          <Form.Item name="city" label="城市" rules={[{ required: true }]}>
            <Input placeholder="请输入城市" />
          </Form.Item>
          <Form.Item name="rankType" label="排名类型" rules={[{ required: true }]}>
            <Select placeholder="请选择排名类型">
              <Option value={1}>类型1</Option>
              <Option value={2}>类型2</Option>
            </Select>
          </Form.Item>
          <Form.Item name="keywords" label="关键词">
            <Select
              mode="tags"
              placeholder="请输入关键词"
              style={{ width: 300 }}
            />
          </Form.Item>
          <Form.Item name="dateRange" label="日期范围">
            <RangePicker format="YYYY-MM-DD" />
          </Form.Item>
          <Form.Item name="displayMode" label="显示模式">
            <Radio.Group>
              <Radio value="single">单关键词</Radio>
              <Radio value="comparison">对比分析</Radio>
            </Radio.Group>
          </Form.Item>
          <Form.Item>
            <Button type="primary" htmlType="submit" loading={loading}>
              查询分析
            </Button>
            <Button style={{ marginLeft: 8 }} onClick={() => form.resetFields()}>
              重置
            </Button>
          </Form.Item>
        </Form>
      </Card>

      {/* 分析结果 */}
      {analysisData && (
        <>
          {/* 汇总信息 */}
          <Card title="分析汇总" style={{ marginBottom: 20 }}>
            <Row gutter={20}>
              <Col span={6}>
                <div style={{ textAlign: 'center' }}>
                  <div style={{ color: '#666', marginBottom: 8 }}>总天数</div>
                  <div style={{ fontSize: 18, fontWeight: 'bold' }}>
                    {analysisData.summary.totalDays} 天
                  </div>
                </div>
              </Col>
              <Col span={6}>
                <div style={{ textAlign: 'center' }}>
                  <div style={{ color: '#666', marginBottom: 8 }}>关键词数量</div>
                  <div style={{ fontSize: 18, fontWeight: 'bold' }}>
                    {analysisData.summary.keywords?.length || 0} 个
                  </div>
                </div>
              </Col>
              <Col span={6}>
                <div style={{ textAlign: 'center' }}>
                  <div style={{ color: '#666', marginBottom: 8 }}>分析位置数</div>
                  <div style={{ fontSize: 18, fontWeight: 'bold' }}>
                    {analysisData.summary.totalPositions} 个
                  </div>
                </div>
              </Col>
              <Col span={6}>
                <div style={{ textAlign: 'center' }}>
                  <div style={{ color: '#666', marginBottom: 8 }}>城市</div>
                  <div style={{ fontSize: 18, fontWeight: 'bold' }}>
                    {analysisData.summary.city}
                  </div>
                </div>
              </Col>
            </Row>
          </Card>

          {/* 矩阵数据 */}
          <Card title="排名矩阵" style={{ marginBottom: 20 }}>
            {Object.keys(analysisData.matrix.keywordMatrices).length === 1 ? (
              renderMatrix(
                Object.values(analysisData.matrix.keywordMatrices)[0],
                analysisData.matrix.headers
              )
            ) : (
              <Tabs>
                {Object.entries(analysisData.matrix.keywordMatrices).map(([keyword, matrix]) => (
                  <Tabs.TabPane tab={keyword} key={keyword}>
                    {renderMatrix(matrix, analysisData.matrix.headers)}
                  </Tabs.TabPane>
                ))}
              </Tabs>
            )}
          </Card>

          {/* 统计信息 */}
          <Row gutter={20}>

                  dataSource={Object.values(analysisData.statistics.keywordCoverage || {})}
                  rowKey="keyword"
                  pagination={false}
                  columns={[
                    { title: '关键词', dataIndex: 'keyword', key: 'keyword' },
                    { title: '覆盖位置数', dataIndex: 'coveredPositions', key: 'coveredPositions' },

                      dataIndex: 'coverageRate',
                      key: 'coverageRate',

                          percent={Math.round(rate * 100)}

                        />
                      )
                    }

        </>
      )}

  )
```

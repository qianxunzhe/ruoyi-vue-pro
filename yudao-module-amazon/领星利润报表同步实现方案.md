# 领星利润报表同步实现方案

## 一、需求背景

### 1.1 业务需求
- 同步领星利润报表数据到本地数据库
- 支持手动同步（前端传入开始和结束时间）
- 查询时支持多维度实时聚合（SKU、SPU、店铺维度）
- 预留SPU字段支持后续扩展
- 使用PostgreSQL数据库

### 1.2 API限制
- 按天查询最多31天跨度
- 单次查询最多返回10000条
- API限流：令牌桶容量10

### 3.2 同步服务实现

```java
package cn.iocoder.yudao.module.amazon.service.profitreport;

import cn.iocoder.yudao.module.amazon.dal.dataobject.profitreport.ProfitReportDO;
import cn.iocoder.yudao.module.amazon.dal.mysql.profitreport.ProfitReportMapper;
import cn.iocoder.yudao.module.amazon.service.lingXingAPI.LingXingApiService;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;

@Service
@Slf4j
public class ProfitReportSyncServiceImpl implements ProfitReportSyncService {
    
    @Autowired
    private LingXingApiService lingXingApiService;
    
    @Autowired
    private ProfitReportMapper profitReportMapper;
    
    @Value("${lingxing.sync.batch-size:1000}")
    private int batchSize;
    
    @Override
    @Async("profitReportSyncExecutor")
    @Transactional
    public void syncProfitReport(LocalDate startDate, LocalDate endDate, SyncType syncType) {
        doSync(startDate, endDate, syncType);
    }
    
    private void doSync(LocalDate startDate, LocalDate endDate, SyncType syncType) {
        ProfitReportSyncLog syncLog = createSyncLog(startDate, endDate, syncType);
        
        try {
            log.info("开始同步利润报表数据: {} 至 {}, 类型: {}", startDate, endDate, syncType);
            
            // 1. 分批获取数据（处理31天限制）
            List<ProfitReportDTO> allData = fetchDataInBatches(startDate, endDate);
            
            // 2. 数据转换
            List<ProfitReportDO> entities = convertToEntities(allData, startDate, endDate);
            
            // 3. 根据同步类型处理
            if (syncType == SyncType.FULL) {
                // 删除旧数据（相同时间范围）
                deleteOldData(startDate, endDate);
            }
            
            // 4. 批量保存
            saveInBatches(entities);
            
            // 5. 更新同步日志
            updateSyncLogSuccess(syncLog, entities.size());
            
            log.info("同步完成，共同步 {} 条数据", entities.size());
            
        } catch (Exception e) {
            log.error("同步失败: syncType={}, dateRange={}-{}", syncType, startDate, endDate, e);
            updateSyncLogFailed(syncLog, e.getMessage());
            throw new BusinessException("同步失败", e);
        }
    }
    
    /**
     * 分批获取数据（处理API的31天限制）
     */
    private List<ProfitReportDTO> fetchDataInBatches(LocalDate startDate, LocalDate endDate) {
        List<ProfitReportDTO> allData = new ArrayList<>();
        LocalDate currentStart = startDate;
        
        while (!currentStart.isAfter(endDate)) {
            // 计算批次结束日期（最多31天）
            LocalDate batchEnd = currentStart.plusDays(30);
            if (batchEnd.isAfter(endDate)) {
                batchEnd = endDate;
            }
            
            // 构建查询参数
            ProfitReportQueryDTO query = ProfitReportQueryDTO.builder()
                .startDate(currentStart.toString())
                .endDate(batchEnd.toString())
                .monthlyQuery(false)
                .offset(0)
                .length(10000)
                .build();
            
            // 分页获取数据
            int offset = 0;
            while (true) {
                query.setOffset(offset);
                ProfitReportResponseDTO response = lingXingApiService.queryProfitReport(query);
                
                if (response.getRecords().isEmpty()) {
                    break;
                }
                
                allData.addAll(response.getRecords());
                offset += response.getRecords().size();
                
                // 如果返回数量小于最大值，说明已经获取完所有数据
                if (response.getRecords().size() < 10000) {
                    break;
                }
            }
            
            currentStart = batchEnd.plusDays(1);
        }
        
        return allData;
    }
    
    /**
     * 转换实体，添加同步时间范围
     */
    private List<ProfitReportDO> convertToEntities(
            List<ProfitReportDTO> dtos, 
            LocalDate startDate, 
            LocalDate endDate) {
        
        return dtos.stream().map(dto -> {
            ProfitReportDO entity = new ProfitReportDO();
            
            // 设置同步时间范围
            entity.setSyncStartDate(startDate);
            entity.setSyncEndDate(endDate);
            
            // 映射基础字段
            entity.setReportId(dto.getId());
            entity.setPostedDate(LocalDate.parse(dto.getPostedDateLocale()));
            entity.setAsin(dto.getAsin());
            entity.setParentAsin(dto.getParentAsin());
            entity.setSid(dto.getSid());
            entity.setLocalSku(dto.getLocalSku());
            entity.setSpu(dto.getSpu());  // 映射SPU字段（如果接口有返回）
            entity.setStoreName(dto.getStoreName());
            entity.setCountryCode(dto.getCountryCode());
            
            // 映射销售数据
            entity.setTotalSalesQuantity(dto.getTotalSalesQuantity());
            entity.setTotalSalesAmount(dto.getTotalSalesAmount());
            entity.setFbaSalesQuantity(dto.getFbaSalesQuantity());
            entity.setFbaSaleAmount(dto.getFbaSaleAmount());
            
            // 映射广告数据
            entity.setTotalAdsSales(dto.getTotalAdsSales());
            entity.setTotalAdsCost(dto.getTotalAdsCost());
            
            // 映射成本和利润
            entity.setCgPriceTotal(dto.getCgPriceTotal());
            entity.setGrossProfit(dto.getGrossProfit());
            entity.setGrossRate(dto.getGrossRate());
            
            // 映射其他重要字段
            entity.setPrincipalRealname(dto.getPrincipalRealname());
            entity.setProductDeveloperRealname(dto.getProductDeveloperRealname());
            entity.setCategoryName(dto.getCategoryName());
            entity.setBrandName(dto.getBrandName());
            entity.setOtherFeeStr(JSON.toJSONString(dto.getOtherFeeStr()));
            
            return entity;
        }).collect(Collectors.toList());
    }
    
    /**
     * 删除相同时间范围的旧数据
     */
    private void deleteOldData(LocalDate startDate, LocalDate endDate) {
        int deleted = profitReportMapper.deleteByDateRange(startDate, endDate);
        log.info("删除旧数据 {} 条", deleted);
    }
    
    /**
     * 批量保存数据
     */
    private void saveInBatches(List<ProfitReportDO> entities) {
        if (entities.isEmpty()) {
            return;
        }
        
        for (int i = 0; i < entities.size(); i += batchSize) {
            int end = Math.min(i + batchSize, entities.size());
            List<ProfitReportDO> batch = entities.subList(i, end);
            profitReportMapper.batchInsert(batch);
            log.debug("保存批次数据: {}-{}/{}", i, end, entities.size());
        }
    }
}
```

### 3.3 业务层聚合服务

```java
package cn.iocoder.yudao.module.amazon.service.profitreport;

import cn.iocoder.yudao.framework.common.pojo.PageResult;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.*;
import java.util.stream.Collectors;

@Service
@Slf4j
public class ProfitReportAggregationService {
    
    @Autowired
    private ProfitReportMapper profitReportMapper;
    
    /**
     * SKU维度聚合
     */
    public PageResult<ProfitReportSkuAggVO> aggregateBySku(ProfitReportQueryVO query) {
        // 1. 从数据库查询原始数据
        List<ProfitReportDO> rawData = profitReportMapper.selectByCondition(query);
        
        // 2. 在内存中按SKU分组聚合
        Map<String, List<ProfitReportDO>> groupedData = rawData.stream()
            .filter(item -> StringUtils.isNotBlank(item.getLocalSku()))
            .collect(Collectors.groupingBy(ProfitReportDO::getLocalSku));
        
        // 3. 计算聚合数据
        List<ProfitReportSkuAggVO> aggregatedList = groupedData.entrySet().stream()
            .map(entry -> {
                String sku = entry.getKey();
                List<ProfitReportDO> skuData = entry.getValue();
                
                ProfitReportSkuAggVO aggVO = new ProfitReportSkuAggVO();
                aggVO.setLocalSku(sku);
                aggVO.setLocalName(skuData.get(0).getLocalName());
                
                // 汇总销售数量
                aggVO.setTotalSalesQuantity(
                    skuData.stream().mapToInt(ProfitReportDO::getTotalSalesQuantity).sum()
                );
                
                // 汇总销售金额
                aggVO.setTotalSalesAmount(
                    skuData.stream()
                        .map(ProfitReportDO::getTotalSalesAmount)
                        .reduce(BigDecimal.ZERO, BigDecimal::add)
                );
                
                // 汇总广告费用
                aggVO.setTotalAdsCost(
                    skuData.stream()
                        .map(ProfitReportDO::getTotalAdsCost)
                        .reduce(BigDecimal.ZERO, BigDecimal::add)
                );
                
                // 汇总成本
                aggVO.setTotalCost(
                    skuData.stream()
                        .map(ProfitReportDO::getTotalCost)
                        .reduce(BigDecimal.ZERO, BigDecimal::add)
                );
                
                // 汇总毛利
                aggVO.setGrossProfit(
                    skuData.stream()
                        .map(ProfitReportDO::getGrossProfit)
                        .reduce(BigDecimal.ZERO, BigDecimal::add)
                );
                
                // 计算毛利率
                if (aggVO.getTotalSalesAmount().compareTo(BigDecimal.ZERO) > 0) {
                    aggVO.setGrossRate(
                        aggVO.getGrossProfit()
                            .divide(aggVO.getTotalSalesAmount(), 4, RoundingMode.HALF_UP)
                            .multiply(new BigDecimal("100"))
                    );
                } else {
                    aggVO.setGrossRate(BigDecimal.ZERO);
                }
                
                // 计算ACOS（广告成本占销售比）
                BigDecimal totalAdsSales = skuData.stream()
                    .map(ProfitReportDO::getTotalAdsSales)
                    .reduce(BigDecimal.ZERO, BigDecimal::add);
                    
                if (totalAdsSales.compareTo(BigDecimal.ZERO) > 0) {
                    aggVO.setAcos(
                        aggVO.getTotalAdsCost()
                            .divide(totalAdsSales, 4, RoundingMode.HALF_UP)
                            .multiply(new BigDecimal("100"))
                    );
                } else {
                    aggVO.setAcos(BigDecimal.ZERO);
                }
                
                // 统计天数
                aggVO.setDaysCount(
                    skuData.stream()
                        .map(ProfitReportDO::getPostedDate)
                        .distinct()
                        .count()
                );
                
                return aggVO;
            })
            .sorted((a, b) -> b.getTotalSalesAmount().compareTo(a.getTotalSalesAmount()))
            .collect(Collectors.toList());
        
        // 4. 内存分页
        return paginateInMemory(aggregatedList, query);
    }
    
    /**
     * SPU维度聚合
     */
    public PageResult<ProfitReportSpuAggVO> aggregateBySpu(ProfitReportQueryVO query) {
        // 1. 查询原始数据
        List<ProfitReportDO> rawData = profitReportMapper.selectByCondition(query);
        
        // 2. 按SPU分组（优先使用spu字段，其次parent_asin，最后asin）
        Map<String, List<ProfitReportDO>> groupedData = rawData.stream()
            .collect(Collectors.groupingBy(item -> {
                // 优先使用SPU字段
                if (StringUtils.isNotBlank(item.getSpu())) {
                    return item.getSpu();
                }
                // 其次使用parent_asin
                if (StringUtils.isNotBlank(item.getParentAsin())) {
                    return item.getParentAsin();
                }
                // 最后使用asin
                return item.getAsin();
            }));
        
        // 3. 聚合计算
        List<ProfitReportSpuAggVO> aggregatedList = groupedData.entrySet().stream()
            .map(entry -> buildSpuAggVO(entry.getKey(), entry.getValue()))
            .sorted((a, b) -> b.getTotalSalesAmount().compareTo(a.getTotalSalesAmount()))
            .collect(Collectors.toList());
        
        return paginateInMemory(aggregatedList, query);
    }
    
    /**
     * 店铺维度聚合
     */
    public PageResult<ProfitReportShopAggVO> aggregateByShop(ProfitReportQueryVO query) {
        // 1. 查询原始数据
        List<ProfitReportDO> rawData = profitReportMapper.selectByCondition(query);
        
        // 2. 按店铺ID分组
        Map<String, List<ProfitReportDO>> groupedData = rawData.stream()
            .collect(Collectors.groupingBy(ProfitReportDO::getSid));
        
        // 3. 聚合计算
        List<ProfitReportShopAggVO> aggregatedList = groupedData.entrySet().stream()
            .map(entry -> {
                String sid = entry.getKey();
                List<ProfitReportDO> shopData = entry.getValue();
                
                ProfitReportShopAggVO aggVO = new ProfitReportShopAggVO();
                aggVO.setSid(sid);
                aggVO.setStoreName(shopData.get(0).getStoreName());
                aggVO.setCountryCode(shopData.get(0).getCountryCode());
                
                // 汇总销售额
                aggVO.setTotalSalesAmount(
                    shopData.stream()
                        .map(ProfitReportDO::getTotalSalesAmount)
                        .reduce(BigDecimal.ZERO, BigDecimal::add)
                );
                
                // 汇总销售量
                aggVO.setTotalSalesQuantity(
                    shopData.stream()
                        .mapToInt(ProfitReportDO::getTotalSalesQuantity)
                        .sum()
                );
                
                // 汇总毛利
                aggVO.setGrossProfit(
                    shopData.stream()
                        .map(ProfitReportDO::getGrossProfit)
                        .reduce(BigDecimal.ZERO, BigDecimal::add)
                );
                
                // 计算毛利率
                if (aggVO.getTotalSalesAmount().compareTo(BigDecimal.ZERO) > 0) {
                    aggVO.setGrossRate(
                        aggVO.getGrossProfit()
                            .divide(aggVO.getTotalSalesAmount(), 4, RoundingMode.HALF_UP)
                            .multiply(new BigDecimal("100"))
                    );
                }
                
                // 统计SKU和ASIN数量
                aggVO.setSkuCount(
                    shopData.stream()
                        .map(ProfitReportDO::getLocalSku)
                        .filter(Objects::nonNull)
                        .distinct()
                        .count()
                );
                
                aggVO.setAsinCount(
                    shopData.stream()
                        .map(ProfitReportDO::getAsin)
                        .filter(Objects::nonNull)
                        .distinct()
                        .count()
                );
                
                return aggVO;
            })
            .sorted((a, b) -> b.getTotalSalesAmount().compareTo(a.getTotalSalesAmount()))
            .collect(Collectors.toList());
        
        return paginateInMemory(aggregatedList, query);
    }
    
    /**
     * 内存分页
     */
    private <T> PageResult<T> paginateInMemory(List<T> list, ProfitReportQueryVO query) {
        int total = list.size();
        int pageNo = query.getPageNo();
        int pageSize = query.getPageSize();
        
        int offset = (pageNo - 1) * pageSize;
        int limit = Math.min(offset + pageSize, total);
        
        if (offset >= total) {
            return new PageResult<>(Collections.emptyList(), 0L);
        }
        
        List<T> pageData = list.subList(offset, limit);
        return new PageResult<>(pageData, (long) total);
    }
}
```

```java
package cn.iocoder.yudao.module.amazon.controller.admin.profitreport;

import javax.validation.Valid;
import java.time.LocalDate;

@Tag(name = "利润报表管理")
@RestController
@RequestMapping("/amazon/profit-report")
public class ProfitReportController {
    
    @Autowired
    private ProfitReportSyncService syncService;

    private ProfitReportAggregationService aggregationService;
    
    @Autowired
    private ProfitReportQueryService queryService;
    
    /**
     * 同步利润报表数据
     */

        // 验证时间范围

            return CommonResult.error("开始时间不能晚于结束时间");
        }
        
        // 异步执行同步任务
        syncService.syncProfitReport(req.getStartDate(), req.getEndDate(), req.getSyncType());
        
        return CommonResult.success(true);
    }
```

## 五、核心特性

### 5.1 数据同步策略
- **手动同步**：前端传入时间范围，支持任意时间段
- **同步类型**：支持全量（FULL）和增量（INCREMENT）同步
- **批量处理**：自动处理API的31天限制，分批获取数据
- **去重机制**：使用联合唯一键（sync_start_date + sync_end_date + asin + sid + posted_date）避免重复

### 5.2 聚合方案
- **实时聚合**：查询时在业务层内存中实时计算
- **多维度支持**：SKU、SPU、店铺三个维度
- **SPU处理**：预留SPU字段，支持后续扩展
- **性能优化**：使用Stream API并行处理，内存分页减少数据库压力

### 5.3 扩展性设计
- **同步类型**：支持全量和增量同步
- **配置化**：通过配置文件调整批量大小等参数
- **异步处理**：同步任务异步执行，避免阻塞主线程
